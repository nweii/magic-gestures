# This is the configuration file for Magic Gestures, which turns Magic Mouse
# and Magic Trackpad gestures into shortcuts, actions, or custom URLs.
#
# A # at the start of a line or after whitespace starts a comment that runs to
# the end of the line. Every other line is live.
#
# Save this file, then pick "Reload Settings" from the menu bar to apply it.
#
# To hide the menu bar icon, use System Settings > Menu Bar. Gestures keep
# working, and you can manage this file from a terminal or an agent.
#
# You can also pick "Manage with Agent" from the menu bar for help configuring,
# updating, troubleshooting, or uninstalling the app. AGENTS.md beside this file
# tells the agent how.
#
# Project: https://github.com/nweii/magic-gestures

[mouse]
hold-right-tap-left = return

[trackpad]
hold-right-tap-left = return

[general]
config-version = 1
enable-mouse = true
enable-trackpad = true
tap-speed = 0.25
verbose-logging = false


# Examples
# ───────────────────────────────────────────────────────────────
#
# Put related bindings under [mouse] or [trackpad] when editing by hand. An
# agent or script can append one binding as mouse.{gesture} or
# trackpad.{gesture}, which overrides the surrounding section. The examples use
# that append-safe form so you can uncomment one where it is.
#
# URL substitutions resolve when the gesture fires. Use
# {{clipboard|urlencode}} for clipboard text in a query parameter. Date and time
# substitutions use Apple date formats and the Mac's local time.
# An agent can help construct an app deep link and save it as a URL binding.
#
# mouse.hold-left-tap-right = cmd+shift+a
# mouse.two-finger-tap = middle-click
# mouse.front-right-tap = escape
# mouse.hold-left-slide-right = url:raycast://extensions/raycast/raycast-ai/ai-chat
# mouse.three-finger-tap = url:things:///add?title={{clipboard|urlencode}}
# trackpad.hold-left-tap-right = mission-control
# trackpad.three-finger-swipe-up = ctrl+up
# trackpad.four-finger-tap = url:things:///add?title={{clipboard|urlencode}}&when={{datetime:yyyy-MM-dd}}
#
# Every gesture, key, and setting:
# https://github.com/nweii/magic-gestures/blob/main/GESTURES.md
