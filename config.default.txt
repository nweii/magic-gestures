# This is the configuration file for Magic Gestures, which turns Magic Mouse
# and Magic Trackpad gestures into shortcuts, actions, custom URLs, or scripts.
#
# A # at the start of a line or after whitespace starts a comment that runs to
# the end of the line. Every other line is live.
#
# Save this file, then pick "Reload Settings" from the menu bar to apply it.
#
# To hide the menu bar icon, use System Settings > Menu Bar. Gestures keep
# working, and you can manage this file from a terminal or an agent.
#
# You can also pick "Manage with Agent" from the menu bar for help configuring,
# updating, troubleshooting, or uninstalling the app. AGENTS.md beside this file
# tells the agent how.
#
# Project: https://github.com/nweii/magic-gestures

[mouse]
hold-left-tap-right = return
hold-right-tap-left = escape
two-finger-tap = middle-click

[trackpad]
hold-left-tap-right = return
hold-right-tap-left = escape

[general]
config-version = 2
enable-mouse = true
enable-trackpad = true
dominant-hand = right
tap-speed = 0.25
haptic-feedback = true
verbose-logging = false


# Examples
# ───────────────────────────────────────────────────────────────
#
# Put bindings under [mouse] or [trackpad]. A section may appear more than once,
# so an agent can append a complete section without depending on the one above.
#
# URL substitutions resolve when the gesture fires. Use
# {{clipboard|urlencode}} for clipboard text in a query parameter. Date and time
# substitutions use Apple date formats and the Mac's local time.
# An agent can help construct an app deep link and save it as a URL binding.
# Use an expanded binding with defer = true when a tap overlaps with a
# double-tap gesture. A second matching tap cancels the delayed action.
#
