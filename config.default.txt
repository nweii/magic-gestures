# MagicGestures
#
# Write `setting = value`. A # starts a comment. Unknown settings are ignored.
#
# A value is either a keystroke, like `return` or `cmd+shift+a`, or a built-in
# action, like `middle-click`. GESTURES.md lists every gesture you can bind and
# every key you can send.
#
# This file is copied to config.txt on first run. Make binding changes in
# config.txt.


[mouse]
# Hold one finger still and tap beside it with another. This shape prevents a
# resting hand from triggering the gesture.
hold-middle-tap-index = return
# hold-index-tap-middle = cmd+shift+a
# two-finger-tap = middle-click


[trackpad]
# These gestures are named for the tap location instead of the tapping finger.
hold-tap-left = return
# hold-tap-right = escape
# three-finger-swipe-up = ctrl+up


[general]
show-menu-bar-icon = true
open-at-login = true

enable-mouse = true
enable-trackpad = true

# Sets the maximum tap duration in seconds. Increase it if taps are not recognized.
tap-speed = 0.25

# Logs every gesture and keystroke to Console.
verbose-logging = false
