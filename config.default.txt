# This is the configuration file for MagicGestures, which turns Magic Mouse
# and Magic Trackpad gestures into keystrokes.
#
# A # starts a comment that runs to the end of the line. Every other line
# is live.
#
# Save this file, then pick Reload Settings from the menu bar to apply it.
#
# Project: https://github.com/nweii/magic-gestures

[mouse]
hold-middle-tap-index = return

[trackpad]
hold-tap-left = return

[general]
show-menu-bar-icon = true
open-at-login = true
enable-mouse = true
enable-trackpad = true
tap-speed = 0.25
verbose-logging = false


# Examples
# ───────────────────────────────────────────────────────────────
#
# You can assign a gesture to a device two ways: put it under [mouse] or
# [trackpad], or write it as mouse.{gesture} or trackpad.{gesture}. These
# use the second form, so you can uncomment one right where it is.
#
# mouse.hold-index-tap-middle = cmd+shift+a
# mouse.two-finger-tap = middle-click
# mouse.front-right-tap = escape
# trackpad.hold-tap-right = mission-control
# trackpad.three-finger-swipe-up = ctrl+up
#
# Every gesture, key, and setting:
# https://github.com/nweii/magic-gestures/blob/main/GESTURES.md
